# If the review app and pipeline live on one box, a plain cron entry is enough.
# Runs daily at 6:00 AM; flock stops overlapping runs.
0 6 * * * cd /opt/bellhaven-sync && flock -n /tmp/bellhaven-sync.lock .venv/bin/python pipeline.py >> logs/pipeline.log 2>&1
